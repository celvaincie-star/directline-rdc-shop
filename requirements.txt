import os
import logging
import base64
import requests
from telegram import Update
from telegram.ext import ApplicationBuilder, CommandHandler, MessageHandler, filters, ContextTypes
from groq import Groq

logging.basicConfig(format='%(asctime)s - %(name)s - %(levelname)s - %(message)s', level=logging.INFO)

TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN")
GROQ_API_KEY = os.getenv("GROQ_API_KEY")

groq_client = Groq(api_key=GROQ_API_KEY)

SYSTEM_PROMPT = """
Tu es l'assistant IA officiel de Directline RDC Shop.
Analyse le texte ou l'image du produit (Taobao, 1688, Pinduoduo, etc.).

Règles de calcul :
- Taux : 7 ¥ = 1 USD.
- Expédition : 25 USD / kg.
- Commission : 10%.

Extrais : Nom, Prix fournisseur en ¥, quantité. Calcule l'estimation en USD. Si le prix est illisible, indique-le.
"""

def encoder_image_depuis_url(file_url):
    response = requests.get(file_url)
    return base64.b64encode(response.content).decode('utf-8')

async def gerer_photo(update: Update, context: ContextTypes.DEFAULT_TYPE):
    msg_attente = await update.message.reply_text("🔎 Analyse de la capture en cours...")
    
    try:
        # Récupérer la plus grande résolution de la photo
-        photo_file = await update.message.photo[-1].get_file()
        image_base64 = encoder_image_depuis_url(photo_file.file_path)
        
        completion = groq_client.chat.completions.create(
            model="llama-3.2-90b-vision-preview",
            messages=[
                {
                    "role": "user",
                    "content": [
                        {"type": "text", "text": SYSTEM_PROMPT},
                        {
                            "type": "image_url",
                            "image_url": {
                                "url": f"data:image/jpeg;base64,{image_base64}"
                            },
                        },
                    ],
                }
            ],
            temperature=0.2,
        )
        reponse_ia = completion.choices[0].message.content
        await msg_attente.edit_text(reponse_ia, parse_mode="Markdown")
    except Exception as e:
        logging.error(f"Erreur Vision : {e}")
        await msg_attente.edit_text("⚠️ Je n'ai pas pu analyser cette image. Essayez d'envoyer le lien textuel.")

if __name__ == '__main__':
    app = ApplicationBuilder().token(TELEGRAM_TOKEN).build()
    app.add_handler(MessageHandler(filters.PHOTO, gerer_photo))
    app.run_polling()
